{{- /* YouTube embed — ported from the hugo-clarity theme. {{< youtube dQw4w9WgXcQ >}} Accepts a bare video ID, or any of the URL shapes YouTube hands out (watch?v=, youtu.be/, /embed/, /shorts/). Responsive 16:9 via aspect-ratio so it needs no wrapper height hack. */ -}} {{- $raw := trim (or (.Get 0) (.Get "id") (.Get "url") "") " " -}} {{- $id := "" -}} {{- if or (hasPrefix $raw "http://") (hasPrefix $raw "https://") -}} {{- $u := urls.Parse $raw -}} {{- if in $u.Host "youtu.be" -}} {{- $id = path.Base $u.Path -}} {{- else if in $u.Host "www.youtube.com" "youtube.com" "m.youtube.com" "music.youtube.com" -}} {{- with $u.Query.Get "v" -}} {{- $id = . -}} {{- else -}} {{- $parts := split $u.Path "/" -}} {{- if in (slice "embed" "shorts" "v" "live") (index $parts 1) -}} {{- $id = index $parts 2 -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $id = $raw -}} {{- end -}} {{- with $id }}
{{- else -}} {{- errorf "[shadcn theme] youtube shortcode: could not parse a video ID from %q" $raw -}} {{- end }}
