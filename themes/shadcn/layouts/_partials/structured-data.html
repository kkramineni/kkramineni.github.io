{{- /* JSON-LD structured data. Projects override this at layouts/_partials/structured-data.html (the shadcn starter does). This minimal version keeps the theme self-contained: it emits WebSite for the site and TechArticle for regular pages, so a site that has not overridden the partial still ships valid structured data rather than nothing at all. Emitted inside one @graph so the article can reference the site by @id instead of duplicating the publisher's name and URL. */ -}} {{- $graph := slice (dict "@type" "WebSite" "@id" (printf "%s#website" site.BaseURL) "url" site.BaseURL "name" site.Title "description" site.Params.description "inLanguage" site.Language.Lang ) -}} {{- if .IsPage -}} {{- $article := dict "@type" "TechArticle" "@id" (printf "%s#article" .Permalink) "headline" (.Title | plainify) "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "inLanguage" site.Language.Lang "isAccessibleForFree" true "publisher" (dict "@id" (printf "%s#website" site.BaseURL)) -}} {{- with .Description -}} {{- $article = merge $article (dict "description" (. | plainify)) -}} {{- end -}} {{- with .Date -}} {{- $article = merge $article (dict "datePublished" (.Format "2006-01-02T15:04:05Z07:00")) -}} {{- end -}} {{- if gt .Lastmod .Date -}} {{- $article = merge $article (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) -}} {{- end -}} {{- $author := .Params.author | default site.Params.author -}} {{- with $author -}} {{- $article = merge $article (dict "author" (dict "@type" "Person" "name" .)) -}} {{- end -}} {{- /* Only emit an image when the page has one — a wrong or missing image is worse than none, since Google flags mismatched structured data. */ -}} {{- with partial "func/cover-image.html" . -}} {{- $article = merge $article (dict "image" (. | absURL)) -}} {{- end -}} {{- with .Params.tags -}} {{- $article = merge $article (dict "keywords" (delimit . ", ")) -}} {{- end -}} {{- $graph = $graph | append $article -}} {{- end -}}
